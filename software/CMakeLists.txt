cmake_minimum_required(VERSION 3.15)
project(mlkem_arm C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(mlkem STATIC src/keccak.c src/poly.c src/mlkem.c src/mlkem_accel.c src/mlkem_rng.c)
target_include_directories(mlkem PUBLIC include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(mlkem PRIVATE -Wall -Wextra -Werror)
endif()
if(UNIX)
  add_executable(mlkem_hps_demo examples/hps_linux_demo.c)
  target_link_libraries(mlkem_hps_demo PRIVATE mlkem)
endif()
